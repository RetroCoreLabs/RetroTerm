    /// <summary>
    /// TDV Keyboard implementation based on TDV-2215 layout (129903)
    /// Handles complete keyboard mapping with proper modifier support
    /// </summary>
    internal class TDVKeyboard
    {
        public delegate void TransmitData(byte value);
        public event TransmitData OnTransmitData;

        // Modifier key states
        private bool lshift = false;
        private bool rshift = false;
        private bool lctrl = false;
        private bool rctrl = false;

        private bool IsShiftDown => lshift || rshift;
        private bool IsCtrlDown => lctrl || rctrl;

        private Dictionary<SDL.SDL_Scancode, KeyboardLayoutEntry> keyboardLayout;

        public TDVKeyboard()
        {
            InitializeKeyboardLayout();
        }

        /// <summary>
        /// Represents a key mapping entry in the TDV-2215 keyboard layout
        /// Based on official layout specification 129903
        /// </summary>
        private class KeyboardLayoutEntry
        {
            /// <summary>Normal key value (unmodified)</summary>
            public byte Normal { get; set; }
            
            /// <summary>Shift key value</summary>
            public byte Shift { get; set; }
            
            /// <summary>Ctrl key value</summary>
            public byte Ctrl { get; set; }
            
            /// <summary>
            /// Control flags determining key behavior
            /// Bit 0: Use with Normal (1=enabled)
            /// Bit 4: Prefix Ctrl with 0xFF (1=yes)
            /// </summary>
            public byte Flags { get; set; }
        }

        internal void KeyDown(SDL.SDL_Keysym keysym)
        {
            HandleKey(keysym, true);
        }

        internal void KeyUp(SDL.SDL_Keysym keysym)
        {
            HandleKey(keysym, false);
        }

        internal void HandleKey(SDL.SDL_Keysym keysym, bool isDown)
        {
            // Handle modifier keys first
            switch (keysym.scancode)
            {
                case SDL.SDL_Scancode.SDL_SCANCODE_LSHIFT:
                    lshift = isDown;
                    return;
                case SDL.SDL_Scancode.SDL_SCANCODE_RSHIFT:
                    rshift = isDown;
                    return;
                case SDL.SDL_Scancode.SDL_SCANCODE_LCTRL:
                    lctrl = isDown;
                    return;
                case SDL.SDL_Scancode.SDL_SCANCODE_RCTRL:
                    rctrl = isDown;
                    return;
            }

            // Only process key presses, not releases (except for modifiers)
            if (!isDown)
                return;

            // Look up the key in our layout
            if (keyboardLayout.TryGetValue(keysym.scancode, out KeyboardLayoutEntry entry))
            {
                ProcessKeyPress(entry);
            }
            else
            {
                Log($"Unmapped key: {keysym.scancode}");
            }
        }

        private void ProcessKeyPress(KeyboardLayoutEntry entry)
        {
            byte codeToTransmit = 0x00;
            bool needsFFPrefix = false;

            // Determine which code to send based on modifier state
            // Priority: Ctrl > Shift > Normal
            if (IsCtrlDown && entry.Ctrl != 0x00)
            {
                codeToTransmit = entry.Ctrl;
                // Check if Ctrl key needs 0xFF prefix (bit 4 of flags)
                needsFFPrefix = (entry.Flags & 0x10) != 0;
            }
            else if (IsShiftDown && entry.Shift != 0x00)
            {
                codeToTransmit = entry.Shift;
            }
            else if ((entry.Flags & 0x01) != 0) // Check if Normal is enabled (bit 0)
            {
                codeToTransmit = entry.Normal;
            }

            // Transmit the key code(s)
            if (codeToTransmit != 0x00)
            {
                if (needsFFPrefix)
                {
                    Transmit(0xFF);
                }
                Transmit(codeToTransmit);
            }
        }

        internal void ProcessCommand(byte value)
        {
            byte cmdGroup = (byte)(value >> 4);
            byte cmdDetail = (byte)(value & 0x0F);

            switch (cmdGroup)
            {
                case 0x00:
                    if (value > 0)
                    {
                        byte led = (byte)((value - 1) & 0x07);
                        bool led_on = ((value >> 3) & 0x01) == 1;
                        SetLed(led, led_on);
                    }
                    break;
                case 0x01:
                    Log($"Turn LED {cmdDetail} on in RAM bitmap only");
                    break;
                case 0x02:
                    ExecuteOperation((byte)(cmdDetail + 1));
                    break;
                default:
                    Log($"Unknown keyboard command 0x{value:X2}");
                    break;
            }
        }

        private void ExecuteOperation(byte operation)
        {
            switch (operation)
            {
                case 1: // Do Nothing
                    Log("Keyboard: Do nothing command");
                    break;
                case 2: // Enable Click
                    Log("Keyboard: Enable Click");
                    break;
                case 3: // Disable Click
                    Log("Keyboard: Disable Click");
                    break;
                case 4: // Turn on CAPS
                    Log("Keyboard: Turn on CAPS");
                    break;
                case 5: // Trigger Beeper
                    Log("Keyboard: Trigger Beeper");
                    break;
                case 6: // Turn off LED 1 to 8
                    Log("Keyboard: Turn off LED 1 to 8");
                    break;
                case 7: // Turn off LED 1 to 4
                    Log("Keyboard: Turn off LED 1 to 4");
                    break;
                case 8: // Check Keyboard Expansion busy state
                    Log("Keyboard: Check Keyboard Expansion busy state");
                    Transmit(0xEF);
                    break;
                case 9: // Enable Key-Repeat
                    Log("Keyboard: Enable Key-Repeat");
                    break;
                case 10: // Disable Key-Repeat
                    Log("Keyboard: Disable Key-Repeat");
                    break;
                case 11: // Read Locks
                    Log("Keyboard: Read locks");
                    // DC: AL1 on, AL2 on
                    // DD: AL1 off, AL2 on  
                    // DE: AL1 on, AL2 off
                    // DF: AL1 off, AL2 off
                    Transmit(0xDC);
                    break;
                case 12: // Disable READY to expansion
                    Log("Keyboard: Disable READY to expansion");
                    break;
                case 13: // Enable READY to expansion
                    Log("Keyboard: Enable READY to expansion");
                    break;
                case 14: // ROM Check (AA = ok, else fail)
                    Log("Keyboard: ROM Check");
                    Transmit(0xAA);
                    break;
                case 15: // RAM Check (AA = ok, EE = fail)
                    Log("Keyboard: RAM Check");
                    Transmit(0xAA);
                    break;
                case 16: // Read Selected Layout ID number (6 chars)
                    Log("Keyboard: Read Selected Layout ID number");
                    // TDV-2215 General Purpose ASCII terminal keyboard
                    // Layout 129903
                    Transmit("129");
                    Transmit("903");
                    break;
                default:
                    Log($"Unknown Keyboard Operation #{operation}");
                    break;
            }
        }

        private void SetLed(byte led, bool led_on)
        {
            Log($"Keyboard LED #{led} {(led_on ? "ON" : "OFF")}");
        }

        private void Transmit(string value)
        {
            foreach (char c in value)
            {
                Transmit((byte)c);
            }
        }

        private void Transmit(byte value)
        {
            if (OnTransmitData != null)
            {
                OnTransmitData(value);
            }
            else
            {
                Log($"Debug: Keyboard Transmit 0x{value:X2} - No OnTransmit()");
            }
        }

        private void Log(string message)
        {
            Logger.Log($"[TDV KEYBOARD] {message}", Logger.LogLevel.Device);
        }

        /// <summary>
        /// Add a character directly (for programmatic input)
        /// </summary>
        internal void AddChar(char key)
        {
            // Convert character to appropriate key code
            byte keyCode = (byte)key;
            
            // Handle special cases
            switch (key)
            {
                case '\r':
                case '\n':
                    Transmit(0xCD); // CR
                    break;
                case '\b':
                    Transmit(0x08); // Backspace
                    break;
                case '\t':
                    Transmit(0x09); // Tab
                    break;
                case ' ':
                    Transmit(0x20); // Space
                    break;
                default:
                    if (key >= 'A' && key <= 'Z')
                    {
                        // Uppercase letter - send as shift
                        Transmit((byte)(0x41 + (key - 'A')));
                    }
                    else if (key >= 'a' && key <= 'z')
                    {
                        // Lowercase letter - send as normal
                        Transmit((byte)(0x61 + (key - 'a')));
                    }
                    else if (key >= '0' && key <= '9')
                    {
                        // Number
                        Transmit((byte)(0x30 + (key - '0')));
                    }
                    else
                    {
                        // Try to send as-is
                        Transmit(keyCode);
                    }
                    break;
            }
        }

        /// <summary>
        /// Send a specific key code directly (for testing/debugging)
        /// </summary>
        internal void SendKeyCode(byte keyCode)
        {
            Transmit(keyCode);
        }

        /// <summary>
        /// Send config menu navigation keys
        /// </summary>
        internal void SendConfigKey(ConfigKey key)
        {
            switch (key)
            {
                case ConfigKey.EnterConfig:
                    Transmit(0xF8); // MODE key (Shift+INSERT)
                    break;
                case ConfigKey.Up:
                    Transmit(0xB1); // Up arrow
                    break;
                case ConfigKey.Down:
                    Transmit(0xB2); // Down arrow
                    break;
                case ConfigKey.Left:
                    Transmit(0xB4); // Left arrow
                    break;
                case ConfigKey.Right:
                    Transmit(0xB3); // Right arrow
                    break;
                case ConfigKey.Enter:
                    Transmit(0xED); // Numeric keypad Enter
                    break;
                case ConfigKey.Exit:
                    Transmit(0xDB); // ESC
                    break;
                case ConfigKey.Home:
                    Transmit(0xB8); // Home
                    break;
            }
        }

        /// <summary>
        /// Configuration menu navigation keys
        /// </summary>
        public enum ConfigKey
        {
            EnterConfig, // 0xF8
            Up,          // 0xB1
            Down,        // 0xB2  
            Left,        // 0xB4
            Right,       // 0xB3
            Enter,       // 0xED
            Exit,        // 0xDB
            Home         // 0xB8
        }
    }
